create database auditoria_db;
use auditoria_db;

create table funcionarios(
	id_func int primary key auto_increment,
    nome varchar(50) not null,
    salario decimal(10,2) not null
);

CREATE TABLE log_salarios (
    id_log          INT AUTO_INCREMENT PRIMARY KEY,
    id_func         INT           NOT NULL,
    salario_antigo  DECIMAL(10,2) NOT NULL,
    salario_novo    DECIMAL(10,2) NOT NULL,
    data_alteracao  DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_log_func FOREIGN KEY (id_func)
        REFERENCES funcionarios (id_func)
);

DELIMITER $$
CREATE FUNCTION fn_calcular_imposto(salario DECIMAL(10,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    IF salario <= 3000.00 THEN
        RETURN 0.00;
    ELSE
        RETURN salario * 0.15;
    END IF;
END $$

CREATE TRIGGER tg_audit_salario
AFTER UPDATE ON funcionarios
FOR EACH ROW
BEGIN
    IF NEW.salario != OLD.salario THEN
        INSERT INTO log_salarios (id_func, salario_antigo, salario_novo, data_alteracao)
        VALUES (OLD.id_func, OLD.salario, NEW.salario, NOW());
    END IF;
END $$
DELIMITER ;
