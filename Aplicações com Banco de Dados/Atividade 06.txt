CREATE DATABASE escolinha_futebol;
USE escolinha_futebol;

CREATE TABLE categoria (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(20) NOT NULL,
    idade_max INT NOT NULL
);

CREATE TABLE atleta (
    id_atleta INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    data_nasc DATE NOT NULL,
    posicao VARCHAR(30),
    telefone_resp VARCHAR(15),
    id_categoria INT NOT NULL,
    FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria)
);

SELECT a.id_atleta, a.nome, DATE_FORMAT(a.data_nasc, '%d/%m/%Y') AS nascimento,
       a.posicao, c.nome AS categoria, a.telefone_resp
FROM atleta a
INNER JOIN categoria c ON c.id_categoria = a.id_categoria
ORDER BY a.nome;
